    ```
    function sm_gmtoffset_init():
        if AutoDetect == false:
            offset_hours = ManualGMT
        else:
            broker_now_utc = TimeGMT()
            broker_server  = TimeCurrent()
            delta_seconds  = broker_server - broker_now_utc
            offset_hours   = round(delta_seconds / 3600)
            if DSTAdjust and is_broker_dst_active():
                offset_hours = offset_hours - 1   # broker time already DST-shifted
        publish_global_variable(GlobalVarName, offset_hours)

    function on_every_tick():
        if minute_of_hour() == 0 and second_of_minute() < 5:
            sm_gmtoffset_init()    # refresh hourly to catch DST switch
    ```

    ```
        if UseGMTOffset:
            broker_offset = read_global_variable("sm_GMTOffset", default=0)

            broker_offset = 0

    function on_new_bar():
        delete_all_objects_with_prefix(ObjectPrefix)
        for day_idx in [-HistoryDays..0]:
            day_anchor = today_midnight_utc + day_idx * 86400
            for session in [Asia, London, US]:
                start_utc = day_anchor + session.start_hour * 3600 + session.start_min * 60
                end_utc   = day_anchor + session.end_hour   * 3600 + session.end_min   * 60
    ```
