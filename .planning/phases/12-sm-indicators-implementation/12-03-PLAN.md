       def compute_pivot_points(df: pd.DataFrame,
                                 params: PivotPointsParams = PivotPointsParams()) -> pd.DataFrame:
           """SM_PivotPoints — standard floor pivots + MMM M1-M4 mid-pivots.

           Per spec Section 5 + MMM Book pp. 42-43. Reads prior-day OHLC via
           shift(1) to prevent lookahead (Pitfall 5). Caller supplies Daily-resampled df.
           """
           out = df.copy()
           prior_h = out["High"].shift(1)
           prior_l = out["Low"].shift(1)
           prior_c = out["Close"].shift(1)
           out["pp"] = (prior_h + prior_l + prior_c) / 3.0
           out["r1"] = 2 * out["pp"] - prior_l

               out["m3"] = (out["pp"] + out["r1"]) / 2.0

       def compute_alerting_tl(df: pd.DataFrame,
                                trendlines: list,  # list of (ts1, p1, ts2, p2)
                                params: AlertingTLParams = AlertingTLParams()) -> pd.DataFrame:
           """SM_Alerting+TL — trendline-touch detector.

           Python port can't read live OBJ_TREND objects; caller supplies
           trendlines explicitly as (start_ts, start_price, end_ts, end_price).
           Each bar: project each TL to bar's time, compute expected price,

           tolerance = params.touch_pips * pip
           out["alert_signal"] = "NONE"
