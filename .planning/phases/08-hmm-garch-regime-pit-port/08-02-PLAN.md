   ```python
   """GARCH(1,1) emission probabilities for HMM-GARCH regime detection.

   Port of V1/helix/src/alpha/regime/emissions.py — verbatim translation.
   No external V1 imports; this file is a leaf module of the regime subpackage.
   """
   from __future__ import annotations

   import math
   from dataclasses import dataclass

   import numpy as np


   @dataclass(frozen=True)
   class GARCHParams:
       """Per-regime GARCH(1,1) parameters: r_t = mu + eps_t,
       eps_t ~ N(0, sigma^2_t),
       sigma^2_t = omega + alpha * eps^2_{t-1} + beta * sigma^2_{t-1}.
       """
       mu: float
       omega: float
       alpha: float
       beta: float

       @property
       def unconditional_variance(self) -> float:
           """Long-run variance: omega / (1 - alpha - beta).

           Only meaningful when self.is_stationary; if alpha+beta >= 1 the
           denominator is non-positive and the value is meaningless.
           """
           return self.omega / (1.0 - self.alpha - self.beta)

       @property
       def is_stationary(self) -> bool:
           """GARCH(1,1) stationarity: alpha + beta < 1."""
           return self.alpha + self.beta < 1.0


   def garch_emission_prob(returns: np.ndarray, params: GARCHParams) -> np.ndarray:
       """Compute log-emission probabilities under a single GARCH(1,1) regime.

       Args:
         returns: array of log-returns, shape (T,).
         params:  fitted GARCHParams (mu, omega, alpha, beta).

       Returns:
         np.ndarray of log-probs shape (T,), values are log f(r_t | regime).

       Implementation:
         eps_t   = r_t - mu
         sigma^2_t = omega + alpha * eps^2_{t-1} + beta * sigma^2_{t-1}
         sigma^2_0 = unconditional_variance
         log b(r_t) = -0.5 * (log(2*pi) + log(sigma^2_t) + eps^2_t / sigma^2_t)
       """
       T = len(returns)
       log_probs = np.empty(T, dtype=np.float64)

       sigma2 = params.unconditional_variance
       eps_prev = 0.0
       sigma2_prev = sigma2
       log_2pi = math.log(2.0 * math.pi)

       for t in range(T):
           eps_t = returns[t] - params.mu
           sigma2 = params.omega + params.alpha * (eps_prev ** 2) + params.beta * sigma2_prev
           if sigma2 <= 0.0:
               # Defensive — should not happen with stationary params and finite returns
               sigma2 = 1e-12
           log_probs[t] = -0.5 * (log_2pi + math.log(sigma2) + (eps_t ** 2) / sigma2)
           eps_prev = eps_t
           sigma2_prev = sigma2

       return log_probs
   ```

   ```python
   """HMM-GARCH two-stage regime detector (Phase 8 / REGM-01 / REGM-02).

   Port of V1/helix/src/alpha/regime/hmm_garch.py with two surgical changes:
     - D-04: predict_viterbi method removed (no Viterbi anywhere in V2).
     - D-04: _compute_log_emission_probs helper removed (only consumer was predict_viterbi).

   Two-stage fit:
     1) GaussianHMM (hmmlearn) on the returns series → state sequence + transmat.
     2) Per-state arch_model GARCH(1,1) → GARCHParams; states sorted ascending
        by unconditional_variance (REGM-02), transmat_/startprob_ remapped.
   """
   from __future__ import annotations

   import logging
   import numpy as np
   from hmmlearn.hmm import GaussianHMM
   from arch import arch_model

   from .emissions import GARCHParams

   logger = logging.getLogger("helix.regime")

   _STATE_LABELS = ("TRENDING", "MEAN_REVERTING", "CRISIS")


   class HMMGARCHRegimeDetector:
       """Two-stage HMM-GARCH regime detector with variance-rank state pinning.

       Public methods:
         fit(returns)            -> bool
         get_regime_label(state) -> str
         is_fitted               -> bool (property)

       Public attributes (after fit):
         garch_params : list[GARCHParams]   # length n_states, ascending variance
         transmat_    : np.ndarray (n_states, n_states)
         startprob_   : np.ndarray (n_states,)

       NOT PROVIDED (per D-04):
         predict_viterbi          — Viterbi decoding banished
         _compute_log_emission_probs — dead helper, only consumer was Viterbi
       """

       def __init__(
           self,
           n_states: int = 3,
           n_iter: int = 100,
           tol: float = 0.01,
           max_retries: int = 5,
           min_state_samples: int = 100,
           random_state: int = 0,
       ) -> None:
           self.n_states = n_states
           self.n_iter = n_iter
           self.tol = tol
           self.max_retries = max_retries
           self.min_state_samples = min_state_samples
           self._random_state = random_state

           self.garch_params: list[GARCHParams] = []
           self.transmat_:  np.ndarray | None = None
           self.startprob_: np.ndarray | None = None
           self._fitted = False

       @property
       def is_fitted(self) -> bool:
           return self._fitted

       def get_regime_label(self, state: int) -> str:
           if 0 <= state < len(_STATE_LABELS):
               return _STATE_LABELS[state]
           raise ValueError(f"Unknown regime state {state}; expected 0..{len(_STATE_LABELS)-1}")

       def fit(self, returns: np.ndarray) -> bool:
           """Two-stage HMM-GARCH fit. Returns True on success, False on failure.

           Failure modes (return False, leave self._fitted = False):
             - GaussianHMM does not converge after max_retries seeds.
             - Any per-state GARCH fit returns non-stationary params (alpha+beta >= 1)
               and the Gaussian fallback also yields non-stationary params.
           """
           returns = np.asarray(returns, dtype=np.float64)
           if returns.ndim != 1:
               raise ValueError(f"returns must be 1-D, got shape {returns.shape}")
           obs = returns.reshape(-1, 1)

           hmm_model = self._fit_gaussian_hmm(obs)
           if hmm_model is None:
               logger.warning("HMMGARCHRegimeDetector: GaussianHMM did not converge")
               return False

           state_seq = hmm_model.predict(obs)

           # Stage 2: per-state GARCH fit
           raw_params: list[tuple[int, GARCHParams]] = []
           for state_idx in range(self.n_states):
               mask = (state_seq == state_idx)
               state_returns = returns[mask]
               if len(state_returns) < self.min_state_samples:
                   params = self._gaussian_fallback(state_returns, state_idx)
               else:
                   params = self._fit_garch(state_returns, state_idx)

               if not params.is_stationary:
                   logger.warning(
                       "HMMGARCHRegimeDetector: state %d non-stationary "
                       "(alpha+beta=%.3f); fit aborted",
                       state_idx, params.alpha + params.beta,
                   )
                   return False

               raw_params.append((state_idx, params))

           # REGM-02: sort by ascending unconditional variance, remap transmat_/startprob_
           raw_params.sort(key=lambda x: x[1].unconditional_variance)
           sort_order = [orig_idx for orig_idx, _ in raw_params]

           self.garch_params = [params for _, params in raw_params]
           self.transmat_  = self._remap_matrix(hmm_model.transmat_,  sort_order)
           self.startprob_ = hmm_model.startprob_[sort_order]
           self._fitted = True
           return True

       def _fit_gaussian_hmm(self, obs: np.ndarray) -> GaussianHMM | None:
           """Fit a GaussianHMM with retry loop. Returns the model or None on failure."""
           last_model: GaussianHMM | None = None
           for attempt in range(self.max_retries):
               seed = self._random_state + attempt
               model = GaussianHMM(
                   n_components=self.n_states,
                   covariance_type="diag",
                   n_iter=self.n_iter,
                   tol=self.tol,
                   random_state=seed,
               )
               try:
                   model.fit(obs)
               except Exception as e:  # noqa: BLE001
                   logger.warning(
                       "GaussianHMM fit attempt %d (seed=%d) raised: %s",
                       attempt, seed, e,
                   )
                   continue
               last_model = model
               if model.monitor_.converged:
                   return model
           return last_model  # may be unconverged but usable for downstream return False

       def _fit_garch(self, state_returns: np.ndarray, state_idx: int) -> GARCHParams:
           """Fit GARCH(1,1) on returns assigned to a single state."""
           res = arch_model(
               state_returns, vol="Garch", p=1, q=1, dist="normal"
           ).fit(disp="off")
           return GARCHParams(
               mu=float(res.params["mu"]),
               omega=float(res.params["omega"]),
               alpha=float(res.params["alpha[1]"]),
               beta=float(res.params["beta[1]"]),
           )

       def _gaussian_fallback(self, state_returns: np.ndarray, state_idx: int) -> GARCHParams:
           """Synthetic GARCHParams when state has too few samples for arch_model."""
           if len(state_returns) > 0:
               mu = float(np.mean(state_returns))
               var = float(np.var(state_returns))
           else:
               mu = 0.0
               var = 1e-8
           omega = max(var * 0.05, 1e-8)
           return GARCHParams(mu=mu, omega=omega, alpha=0.05, beta=0.90)

       @staticmethod
       def _remap_matrix(matrix: np.ndarray, sort_order: list[int]) -> np.ndarray:
           """Re-permute a square matrix by row+column according to sort_order."""
           n = len(sort_order)
           remapped = np.empty((n, n), dtype=matrix.dtype)
           for new_i, old_i in enumerate(sort_order):
               for new_j, old_j in enumerate(sort_order):
                   remapped[new_i, new_j] = matrix[old_i, old_j]
           return remapped
   ```
