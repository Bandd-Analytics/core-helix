   ```python
   """OnlineRegimeFilter — forward-algorithm regime classifier (REGM-01 online half).

   Port of V1/helix/src/alpha/regime/online_filter.py with two surgical changes:
     - DROP dead `garch_emission_prob` import (V1 had it on line 9 but never called).
     - SWAP V1 paths to V2 (.types, .hmm_garch).
   """
   from __future__ import annotations

   import math
   from typing import TYPE_CHECKING

   import numpy as np

   from .types import RegimeState

   if TYPE_CHECKING:
       from .hmm_garch import HMMGARCHRegimeDetector


   class OnlineRegimeFilter:
       """Forward-algorithm online regime filter (D-21 contract)."""

       def __init__(self, detector: "HMMGARCHRegimeDetector") -> None:
           # Local import to break circular reference between hmm_garch and online_filter
           from .hmm_garch import HMMGARCHRegimeDetector  # noqa: PLC0415
           if not isinstance(detector, HMMGARCHRegimeDetector):
               raise TypeError(
                   f"OnlineRegimeFilter requires HMMGARCHRegimeDetector, got {type(detector)}"
               )
           if not detector.is_fitted:
               raise ValueError("OnlineRegimeFilter requires a fitted detector")

           self.detector = detector
           self.n_states = detector.n_states
           self._startprob = np.asarray(detector.startprob_, dtype=np.float64).copy()
           self._transmat  = np.asarray(detector.transmat_,  dtype=np.float64).copy()

           # _alpha: forward-probability vector (filtered state distribution).
           self._alpha = self._startprob.copy()

           # Per-state conditional variance σ²; initialized to unconditional variance.
           self._sigma2 = np.array(
               [p.unconditional_variance for p in detector.garch_params],
               dtype=np.float64,
           )
           # Per-state previous innovation ε_{t-1}; starts at 0.
           self._eps_prev = np.zeros(self.n_states, dtype=np.float64)

           self._log_2pi = math.log(2.0 * math.pi)

       def update(self, return_value: float) -> tuple[RegimeState, float]:
           """One forward step. Returns (RegimeState, confidence float in [0,1])."""
           r = float(return_value)
           b = np.empty(self.n_states, dtype=np.float64)
           new_sigma2 = np.empty(self.n_states, dtype=np.float64)
           new_eps    = np.empty(self.n_states, dtype=np.float64)

           for j, p in enumerate(self.detector.garch_params):
               eps = r - p.mu
               sigma2_j = p.omega + p.alpha * (self._eps_prev[j] ** 2) \
                          + p.beta * self._sigma2[j]
               if sigma2_j <= 0.0:
                   sigma2_j = 1e-12
               b[j] = math.exp(
                   -0.5 * (self._log_2pi + math.log(sigma2_j) + (eps ** 2) / sigma2_j)
               )
               new_sigma2[j] = sigma2_j
               new_eps[j]    = eps

           alpha_new = b * (self._alpha @ self._transmat)
           total = alpha_new.sum()

           if total <= 0.0 or not np.isfinite(total):
               self._sigma2 = new_sigma2
               self._eps_prev = new_eps
               return self._log_space_forward(r)

           self._alpha = alpha_new / total
           self._sigma2 = new_sigma2
           self._eps_prev = new_eps

           idx = int(np.argmax(self._alpha))
           return RegimeState(idx), float(self._alpha[idx])

       def reset(self) -> None:
           """Restore filter state to initial (start-of-stream) values."""
           self._alpha = self._startprob.copy()
           self._sigma2 = np.array(
               [p.unconditional_variance for p in self.detector.garch_params],
               dtype=np.float64,
           )
           self._eps_prev = np.zeros(self.n_states, dtype=np.float64)

       @property
       def state_probs(self) -> np.ndarray:
           """Return a copy of the current forward-probability vector."""
           return self._alpha.copy()

       def _log_space_forward(self, return_value: float) -> tuple[RegimeState, float]:
           """Underflow fallback — V1 lines 125-148 ported."""
           r = float(return_value)
           log_b = np.empty(self.n_states, dtype=np.float64)
           for j, p in enumerate(self.detector.garch_params):
               eps = r - p.mu
               sigma2_j = self._sigma2[j]  # already updated in caller
               log_b[j] = -0.5 * (
                   self._log_2pi + math.log(max(sigma2_j, 1e-300)) + (eps ** 2) / sigma2_j
               )

           with np.errstate(divide="ignore"):
               log_alpha = np.log(np.maximum(self._alpha, 1e-300))
               log_A     = np.log(np.maximum(self._transmat, 1e-300))

           log_alpha_new = log_b + np.array([
               np.logaddexp.reduce(log_alpha + log_A[:, j])
               for j in range(self.n_states)
           ])
           log_total = np.logaddexp.reduce(log_alpha_new)
           self._alpha = np.exp(log_alpha_new - log_total)

           idx = int(np.argmax(self._alpha))
           return RegimeState(idx), float(self._alpha[idx])
   ```

1. Create `V2/v3_intelligence/pit.py` directly from RESEARCH §C.7. Self-contained (only depends on pandas):

   ```python
   """Point-in-time replay clock — runtime no-future-read enforcement (REGM-03).

   Lightweight pandas-native context manager. No ArcticDB dependency. Compares
   read timestamps against an as-of cutoff and raises FutureBarReadError on any
   access where the read timestamp exceeds the cutoff.

   Per CONTEXT.md decisions:
     D-06: replay-clock context manager, pandas-native, no ArcticDB.
     D-07: opt-in via decorator on backtest method (existing loops untouched).
     D-08: timestamp-based check (`bar.ts > as_of_ts`), not index-based.
     D-09: mandatory test — out-of-order read raises FutureBarReadError.
     D-25: PitClock.UNBOUNDED sentinel disables enforcement (offline-fit usage).
   """
   from __future__ import annotations

   from typing import Any, ClassVar, Optional

   import pandas as pd


   class FutureBarReadError(Exception):
       """Raised when a PiT-gated read exceeds the as-of timestamp."""


   class PitClock:
       """Replay clock. Wrap a backtest loop body to enforce no-future reads."""

       UNBOUNDED: ClassVar["PitClock"]   # set after class def — disabled-enforcement sentinel

       def __init__(self, as_of_ts: Optional[pd.Timestamp]) -> None:
           # None marks the UNBOUNDED sentinel; callers should prefer the constant.
           self._as_of: Optional[pd.Timestamp] = as_of_ts
           self._active: bool = False

       def __enter__(self) -> "PitClock":
           self._active = True
           return self

       def __exit__(self, exc_type, exc_val, exc_tb) -> None:
           self._active = False

       def advance(self, new_ts: pd.Timestamp) -> None:
           """Move the cutoff forward inside the loop body. Must be monotone."""
           if self._as_of is not None and new_ts < self._as_of:
               raise ValueError(
                   f"PitClock cannot rewind ({new_ts} < {self._as_of})"
               )
           self._as_of = new_ts

       def read(
           self,
           df: pd.DataFrame,
           sym: Optional[str] = None,
       ) -> pd.DataFrame:
           """Return rows of df whose index is <= self._as_of."""
           if self._as_of is None:   # UNBOUNDED
               return df
           if len(df) == 0:
               raise FutureBarReadError(
                   "Empty DataFrame passed to PitClock.read"
                   + (f" for {sym}" if sym else "")
               )
           if df.index.max() <= self._as_of:
               return df
           truncated = df.loc[df.index <= self._as_of]
           if truncated.empty:
               raise FutureBarReadError(
                   f"No bars at or before {self._as_of}"
                   + (f" for {sym}" if sym else "")
               )
           return truncated

       def assert_no_future(
           self,
           ts: pd.Timestamp,
           sym: Optional[str] = None,
       ) -> None:
           """Explicit guard: raise FutureBarReadError if ts > self._as_of."""
           if self._as_of is None:   # UNBOUNDED
               return
           if ts > self._as_of:
               raise FutureBarReadError(
                   f"Read at {ts} exceeds clock {self._as_of}"
                   + (f" for {sym}" if sym else "")
               )


   # Sentinel construction — single shared instance with as_of=None (D-25).
   PitClock.UNBOUNDED = PitClock(None)


   def pit_gated(method: Any) -> Any:
       """Decorator: backtest method receives a PitClock kw or default UNBOUNDED."""
       def wrapper(self, *args: Any, clock: Optional[PitClock] = None,
                    **kwargs: Any) -> Any:
           if clock is None:
               clock = PitClock.UNBOUNDED
           return method(self, *args, clock=clock, **kwargs)

       wrapper.__name__ = getattr(method, "__name__", "wrapper")
       wrapper.__doc__ = method.__doc__
       return wrapper
   ```

   ```python
   """JSON persistence for fitted HMMGARCHRegimeDetector instances (D-11)."""
   from __future__ import annotations

   import json
   from datetime import datetime, timezone
   from pathlib import Path

   import numpy as np

   from .emissions import GARCHParams
   from .hmm_garch import HMMGARCHRegimeDetector

   SCHEMA_VERSION = 1
   STATE_LABELS = ["TRENDING", "MEAN_REVERTING", "CRISIS"]


   def save_detector(
       detector: HMMGARCHRegimeDetector,
       path: Path,
       *,
       pair: str,
       data_path: str,
       data_window: str = "4yr",
       n_bars: int = 0,
       hmmlearn_converged: bool = True,
       v1_parity_tested: bool = False,
   ) -> None:
       """Serialize a fitted detector to JSON.

       Raises:
         RuntimeError: if detector has not been fit.
       """
       if not detector.is_fitted:
           raise RuntimeError(
               "save_detector: detector has not been fit; cannot persist."
           )
       if detector.transmat_ is None or detector.startprob_ is None:
           raise RuntimeError(
               "save_detector: detector is missing transmat_ or startprob_."
           )

       blob = {
           "schema_version": SCHEMA_VERSION,
           "pair": pair,
           "n_states": detector.n_states,
           "garch_params": [
               {"mu": p.mu, "omega": p.omega, "alpha": p.alpha, "beta": p.beta}
               for p in detector.garch_params
           ],
           "transmat":  np.asarray(detector.transmat_,  dtype=np.float64).tolist(),
           "startprob": np.asarray(detector.startprob_, dtype=np.float64).tolist(),
           "variance_ordering": {
               "state_labels": list(STATE_LABELS[: detector.n_states]),
               "unconditional_variances": [
                   p.unconditional_variance for p in detector.garch_params
               ],
           },
           "fit_metadata": {
               "fitted_at_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
               "data_window":   data_window,
               "data_path":     data_path,
               "n_bars":        int(n_bars),
               "hmmlearn_converged": bool(hmmlearn_converged),
               "v1_parity_tested":   bool(v1_parity_tested),
           },
       }

       Path(path).parent.mkdir(parents=True, exist_ok=True)
       with Path(path).open("w") as f:
           json.dump(blob, f, indent=2)


   def load_detector(path: Path) -> HMMGARCHRegimeDetector:
       """Reconstruct a fitted detector from a JSON file.

       Raises:
         FileNotFoundError: if path does not exist.
         KeyError:          if the JSON is missing schema_version.
         ValueError:        if schema_version is not 1.
       """
       if not Path(path).exists():
           raise FileNotFoundError(f"Detector JSON not found: {path}")

       with Path(path).open("r") as f:
           blob = json.load(f)

       if "schema_version" not in blob:
           raise KeyError(
               f"{path} missing required field 'schema_version'; cannot load."
           )
       if blob["schema_version"] != SCHEMA_VERSION:
           raise ValueError(
               f"{path} schema_version={blob['schema_version']} not supported "
               f"(expected {SCHEMA_VERSION})."
           )

       n_states = int(blob["n_states"])
       det = HMMGARCHRegimeDetector(n_states=n_states)
       det.garch_params = [
           GARCHParams(
               mu=float(p["mu"]), omega=float(p["omega"]),
               alpha=float(p["alpha"]), beta=float(p["beta"]),
           )
           for p in blob["garch_params"]
       ]
       det.transmat_  = np.asarray(blob["transmat"],  dtype=np.float64)
       det.startprob_ = np.asarray(blob["startprob"], dtype=np.float64)
       det._fitted = True
       return det
   ```
