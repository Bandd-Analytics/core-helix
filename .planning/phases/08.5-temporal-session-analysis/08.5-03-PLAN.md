    ```python
    # ── Add to existing imports at top of file ───────────────────────────────
    import seaborn as sns
    import matplotlib.pyplot as plt

    # ── Append after write_combo_csv (~120 lines) ────────────────────────────

    # Frozen render contract (test_heatmap_diverging_colormap inspects this)
    RENDER_KWARGS: dict = {
        "cmap":   "RdYlGn",   # RESEARCH §Pattern 3 — diverging colormap
        "center": 0,          # anchors zero-Sharpe to colormap midpoint
        "vmin":   -1.0,       # clip outliers below (rare bucket Sharpe < -1 dominates colormap otherwise)
        "vmax":    1.0,       # clip outliers above
        "annot":  True,       # numeric annotation on each cell
        "fmt":    ".2f",
    }

    def build_heatmap_mask(bucket_df: pd.DataFrame,
                            min_trades: int = MIN_TRADES) -> pd.Series:
        """Return a boolean Series indexed by the dim column.

        True = mask out (insufficient_evidence, render gray).
        False = render normally.

        Per CONTEXT D-03 + D-04: cells with trade_count < min_trades MUST be masked,
        not zero-filled (RESEARCH §Anti-Patterns: zero-fill lies green/yellow).
        """
        # bucket_df has columns: <dim>, sharpe, win_rate, trade_count, status
        # The dim column name is the first column
        dim_col = bucket_df.columns[0]
        mask = bucket_df.set_index(dim_col)["trade_count"] < min_trades
        return mask

    def render_combo_heatmaps(buckets: dict[str, pd.DataFrame],
                               pair: str, strategy: str, timeframe: str,
                               out_dir: Path) -> list[Path]:
        """Render one PNG per dimension per (pair, strategy, timeframe).

        Per CONTEXT D-14:
          - hour + dow always rendered for every combo
          - dom + doy ONLY rendered for H1 / Daily combos (M15 corpus too thin)
        Per RESEARCH §Pattern 3:
          - cmap='RdYlGn', center=0, vmin/vmax=±1.0
          - mask=trade_count<30 (gray cells)
        Per CONTEXT D-14:
          - Output path: {out_dir}/heatmap_{dim}_{pair}_{strategy}_{timeframe}.png

        Returns list of written PNG paths.
        """
        out_dir.mkdir(parents=True, exist_ok=True)
        written: list[Path] = []

        # Determine which dims to render (M15 skips dom/doy)
        dims_to_render = list(DIMS_ALWAYS)  # session, hour, dow
        if timeframe in ("H1", "Daily"):
            dims_to_render += list(DIMS_H1_DAILY_ONLY)  # dom, doy
        # session is a categorical (string), heatmap renders less cleanly — skip session heatmap
        # session is reported in CSV; heatmaps are for the integer dims (hour/dow/dom/doy)
        dims_to_render = [d for d in dims_to_render if d != "session"]

        for dim in dims_to_render:
            df = buckets.get(dim)
            if df is None or df.empty:
                continue

            # Build matrix: 1 row × N columns (where N = bucket count for this dim)
            matrix = df.set_index(dim)[["sharpe"]].T   # shape (1, N)
            mask_series = build_heatmap_mask(df)        # Series indexed by dim
            mask = mask_series.values.reshape(1, -1)    # shape (1, N), aligned with matrix

            # Figure sizing: scale width with bucket count
            width = max(8, 0.5 * len(df))
            fig, ax = plt.subplots(figsize=(width, 2.5))

            sns.heatmap(
                matrix,
                mask=mask,
                cmap=RENDER_KWARGS["cmap"],
                center=RENDER_KWARGS["center"],
                vmin=RENDER_KWARGS["vmin"],
                vmax=RENDER_KWARGS["vmax"],
                annot=RENDER_KWARGS["annot"],
                fmt=RENDER_KWARGS["fmt"],
                cbar_kws={"label": "Sharpe (annualized, clipped to ±1.0)"},
                ax=ax,
            )
            ax.set_title(f"{pair} / {strategy} / {timeframe} — Sharpe by {dim}")
            ax.set_xlabel(dim)
            ax.set_ylabel("")
            fig.tight_layout()

            out_path = out_dir / f"heatmap_{dim}_{pair}_{strategy}_{timeframe}.png"
            fig.savefig(out_path, dpi=120)
            plt.close(fig)
            written.append(out_path)

        return written
    ```
