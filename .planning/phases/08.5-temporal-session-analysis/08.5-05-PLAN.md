    ```python
    # ── SESS-04: session_config.py code generator ────────────────────────────
    import hashlib
    from datetime import datetime, timezone

    def _build_session_rules(evidence_dir: Path) -> dict[tuple[str, str, str], dict]:
        """Read session_performance_{pair}_{strategy}_{timeframe}.csv files and derive rules.

        Per CONTEXT D-04 + RESEARCH Pattern 6:
          - blacklisted_hours: hours where Sharpe <= SHARPE_BAD AND status != insufficient_evidence
          - blacklisted_dows:  dows  where Sharpe <= SHARPE_BAD AND status != insufficient_evidence
          - tradeable_sessions: sessions where Sharpe >= SHARPE_GOOD AND status != insufficient_evidence

        Buckets in (-0.2, 0.3) generate no rule (neither allow nor block — D-04).
        """
        rules: dict[tuple[str, str, str], dict] = {}
        for csv_path in sorted(evidence_dir.glob("session_performance_*.csv")):
            df = pd.read_csv(csv_path)
            if df.empty: continue
            # Filename: session_performance_{pair}_{strategy}_{timeframe}.csv
            stem = csv_path.stem.removeprefix("session_performance_")
            parts = stem.rsplit("_", 2)  # split from the right
            if len(parts) != 3:
                # parts[0] = pair, parts[1] = strategy, parts[2] = timeframe
                continue
            pair, strategy, timeframe = parts[0], parts[1], parts[2]

            evaluable = df[df["status"] != "insufficient_evidence"].copy()
            evaluable["sharpe"] = pd.to_numeric(evaluable["sharpe"], errors="coerce")

            blacklisted_hours = sorted({
                int(r["bucket"]) for _, r in
                evaluable[(evaluable["dim"] == "hour") & (evaluable["sharpe"] <= SHARPE_BAD)].iterrows()
            })
            blacklisted_dows = sorted({
                int(r["bucket"]) for _, r in
                evaluable[(evaluable["dim"] == "dow") & (evaluable["sharpe"] <= SHARPE_BAD)].iterrows()
            })
            tradeable_sessions = sorted({
                str(r["bucket"]) for _, r in
                evaluable[(evaluable["dim"] == "session") & (evaluable["sharpe"] >= SHARPE_GOOD)].iterrows()
                if str(r["bucket"]) != "OFF"
            })
            if blacklisted_hours or blacklisted_dows or tradeable_sessions:
                rules[(pair, strategy, timeframe)] = {
                    "blacklisted_hours":  blacklisted_hours,
                    "blacklisted_dows":   blacklisted_dows,
                    "tradeable_sessions": tradeable_sessions,
                }
        return rules

    def _read_blackout_patterns(risk_calendar_path: Path) -> list[dict]:
        """Read risk_calendar.yaml and project to BLACKOUT_PATTERNS shape.

        Each pattern dict: pattern, time_utc=(hh,mm), duration_min, affects, source,
        plus pattern-specific fields (n, dow, day, month, dates).
        """
        from ruamel.yaml import YAML
        yaml = YAML(typ="rt")
        if not risk_calendar_path.exists():
            return []
        with risk_calendar_path.open() as f:
            data = yaml.load(f) or {}
        out: list[dict] = []
        for entry in (data.get("blackouts") or []):
            time_str = entry.get("time", "00:00")
            try:
                hh, mm = time_str.split(":")
                hh, mm = int(hh), int(mm)
            except (ValueError, AttributeError):
                hh, mm = 0, 0
            projected = {
                "pattern":      entry.get("pattern", "dates"),
                "time_utc":     (hh, mm),
                "duration_min": int(entry.get("duration_min", 30)),
                "affects":      list(entry.get("affects", [])),
                "source":       entry.get("source", "empirical"),
            }
            for fld in ("n", "dow", "day", "month", "dates"):
                if fld in entry: projected[fld] = entry[fld]
            out.append(projected)
        return out

    def regenerate_session_config(evidence_dir: Path, target: Path,
                                    risk_calendar_path: Path | None = None) -> Path:
        """Generate V2/v3_intelligence/session_config.py from evidence/ artifacts.

        Output is a Python literal file (CONTEXT D-08) consumed by temporal_filters.py.
        Per RESEARCH Pattern 6: matches pair_config.py convention (generated data as literals).
        """
        if risk_calendar_path is None:
            risk_calendar_path = evidence_dir / "risk_calendar.yaml"

        rules    = _build_session_rules(evidence_dir)
        patterns = _read_blackout_patterns(risk_calendar_path)

        generated_at = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
        h = hashlib.sha256()
        for csv_path in sorted(evidence_dir.glob("session_performance_*.csv")):
            h.update(csv_path.read_bytes())
        if risk_calendar_path.exists():
            h.update(risk_calendar_path.read_bytes())
        source_hash = h.hexdigest()[:16]

        body = [
            '"""GENERATED BY V2/scripts/run_temporal_analysis.py — DO NOT EDIT BY HAND.',
            "",
            "Phase 8.5 SESS-04 — emitted from .planning/phases/08.5-temporal-session-analysis/evidence/",
            "Re-generate via: cd V2 && python -m scripts.run_temporal_analysis",
            '"""',
            "from __future__ import annotations",
            "",
            f'GENERATED_AT: str = "{generated_at}"',
            f'SOURCE_HASH:  str = "{source_hash}"',
            "",
            "# Per (pair, strategy, timeframe), the set of buckets that gate entries.",
            "# blacklisted_hours / blacklisted_dows -> hard veto (Sharpe <= -0.2)",
            "# tradeable_sessions -> empirically validated as Sharpe >= 0.3",
            "SESSION_RULES: dict[tuple[str, str, str], dict] = {",
        ]
        for key, rule in sorted(rules.items()):
            pair, strat, tf = key
            body.append(f'    ({pair!r}, {strat!r}, {tf!r}): {{')
            body.append(f'        "blacklisted_hours":  {rule["blacklisted_hours"]!r},')
            body.append(f'        "blacklisted_dows":   {rule["blacklisted_dows"]!r},')
            body.append(f'        "tradeable_sessions": {rule["tradeable_sessions"]!r},')
            body.append(f'    }},')
        body += [
            "}",
            "",
            "# Empirically detected + manually overridden recurring blackouts.",
            "# Each entry resolved at call-time by temporal_filters.is_blackout_window(ts).",
            "BLACKOUT_PATTERNS: list[dict] = [",
        ]
        for p in patterns:
            body.append(f'    {p!r},')
        body += ["]", ""]

        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text("\n".join(body))
        return target
    ```

    ```python
    """Phase 8.5 SESS-04 — predicate API consumed by Phase 9 StrategyRouter.

    Per CONTEXT D-06 / D-07:
      is_tradeable_session(pair, strategy, timeframe, ts) -> bool  (hard veto)
      is_blackout_window(ts) -> bool                                (global ts-only filter)

    Imports SESSION_RULES + BLACKOUT_PATTERNS from session_config (generated by
    run_temporal_analysis.py per RESEARCH Pattern 6).
    """
    from __future__ import annotations
    from datetime import datetime, timedelta

    from .session_config import SESSION_RULES, BLACKOUT_PATTERNS


    def _pair_currencies(pair: str) -> list[str]:
        """USDJPY -> ['USD', 'JPY']."""
        if len(pair) != 6:
            return []
        return [pair[:3], pair[3:]]


    def _matches_pattern(pattern: dict, ts: datetime) -> bool:
        """Resolve a parametric blackout pattern against a concrete UTC timestamp.

        Per CONTEXT D-01 — ts is assumed UTC.
        """
        ptype = pattern.get("pattern", "")
        time_utc = pattern.get("time_utc", (0, 0))
        if not isinstance(time_utc, (tuple, list)) or len(time_utc) != 2:
            return False
        hh, mm = int(time_utc[0]), int(time_utc[1])
        duration = int(pattern.get("duration_min", 30))
        start = ts.replace(hour=hh, minute=mm, second=0, microsecond=0)
        end   = start + timedelta(minutes=duration)
        in_window = start <= ts < end

        if ptype == "wom":
            n_required   = int(pattern.get("n", 0))
            dow_required = int(pattern.get("dow", -1))
            nth_dow = (ts.day - 1) // 7 + 1
            if nth_dow != n_required:    return False
            if ts.weekday() != dow_required: return False
            return in_window

        if ptype == "dow":
            dow_required = int(pattern.get("dow", -1))
            if ts.weekday() != dow_required: return False
            return in_window

        if ptype == "mom":
            day_required = int(pattern.get("day", 0))
            if ts.day != day_required: return False
            return in_window

        if ptype == "yearly":
            m = int(pattern.get("month", 0))
            d = int(pattern.get("day", 0))
            if ts.month != m or ts.day != d: return False
            return in_window

        if ptype == "dates":
            dates = pattern.get("dates", [])
            day_str = ts.strftime("%Y-%m-%d")
            if day_str not in dates: return False
            return in_window

        return False


    def is_tradeable_session(pair: str, strategy: str, timeframe: str,
                              ts: datetime) -> bool:
        """CONTEXT D-06 — hard veto. False blocks entry; True permits (other gates may still reject).

        Logic:
          1. Look up SESSION_RULES[(pair, strategy, timeframe)]; if no rule, default True.
          2. If ts.hour in blacklisted_hours -> False
          3. If ts.weekday() in blacklisted_dows -> False
          4. If any BLACKOUT_PATTERNS matches AND pattern.affects intersects pair currencies -> False
          5. Otherwise True
        """
        rule = SESSION_RULES.get((pair, strategy, timeframe))
        if rule is not None:
            if ts.hour in rule.get("blacklisted_hours", []):
                return False
            if ts.weekday() in rule.get("blacklisted_dows", []):
                return False

        affected = set(_pair_currencies(pair))
        for pattern in BLACKOUT_PATTERNS:
            pat_affects = set(pattern.get("affects", []) or [])
            if pat_affects and not (affected & pat_affects):
                continue
            if _matches_pattern(pattern, ts):
                return False
        return True


    def is_blackout_window(ts: datetime) -> bool:
        """CONTEXT D-07 — global ts-only filter (currency-agnostic)."""
        return any(_matches_pattern(p, ts) for p in BLACKOUT_PATTERNS)
    ```
