    ```python
    # ── Risk calendar Stage A: empirical detection ───────────────────────────

    DETECTION_SIGMA: float     = 2.5    # CONTEXT D-10 — ~99th percentile threshold
    DETECTION_STD_FLOOR: float = 1e-5   # RESEARCH Pitfall 4 — skip low-liquidity baselines

    def detect_blackout_bars(bars: pd.DataFrame,
                              sigma: float = DETECTION_SIGMA) -> pd.DatetimeIndex:
        """Stage A — return timestamps where realized range > sigma*std above
        same-(hour, dow) 4yr baseline.

        Per CONTEXT D-09 + D-10: empirical detection on full 4yr corpus, full-window baseline
        (NOT rolling — rolling smears recurring events into their own baseline per RESEARCH Anti-Patterns).

        Per RESEARCH §Pitfall 4: when bars table lacks `spread`, use range_pct = (High - Low) / Close
        as a realized-vol proxy. Baseline cells with std < DETECTION_STD_FLOOR are skipped to avoid
        false positives at low-liquidity hours.

        Args:
            bars: DataFrame with index=ts (tz-aware UTC), columns including High, Low, Close.
            sigma: z-score threshold (default 2.5 per D-10).

        Returns: DatetimeIndex of timestamps qualifying as detected anomalies.
        """
        if bars is None or len(bars) == 0:
            return pd.DatetimeIndex([])

        df = bars.copy()
        # Realized-range proxy (Title-case OHLC per D-17 / Phase 8.4 D-03)
        df["range_pct"] = (df["High"] - df["Low"]) / df["Close"]
        # Same-(hour, dow) baseline — full-window, NOT rolling
        df["hour"] = df.index.hour
        df["dow"]  = df.index.dayofweek

        baseline = df.groupby(["hour", "dow"])["range_pct"].agg(["mean", "std"]).reset_index()
        # Apply std floor — replace tiny std with NaN so z-score becomes NaN (no detection)
        baseline.loc[baseline["std"] < DETECTION_STD_FLOOR, "std"] = float("nan")

        # Join baseline back onto bars
        merged = df.reset_index().merge(baseline, on=["hour", "dow"], how="left")
        merged["zscore"] = (merged["range_pct"] - merged["mean"]) / merged["std"]
        merged = merged.set_index("ts" if "ts" in merged.columns else merged.columns[0])

        # Detected: zscore > sigma (NaN comparisons return False — std-floor skips)
        detected_mask = merged["zscore"] > sigma
        return pd.DatetimeIndex(merged.index[detected_mask])
    ```

    ```python
    # ── Risk calendar Stage B: pattern clustering ────────────────────────────

    def _pair_currencies(pair: str) -> list[str]:
        """USDJPY -> ['USD', 'JPY']; EURGBP -> ['EUR', 'GBP']."""
        if len(pair) != 6:
            raise ValueError(f"expected 6-char pair, got {pair!r}")
        return [pair[:3], pair[3:]]


    def _matches_wom(stamps: pd.DatetimeIndex) -> tuple[bool, int | None, int | None]:
        """Test if all stamps fall on the Nth occurrence of weekday W in their respective months.

        Returns (matches, n, dow) — (True, n, dow) if all stamps are the Nth occurrence of dow=W;
                                    (False, None, None) otherwise.

        n = (day - 1) // 7 + 1  -> 1, 2, 3, 4 (5th does occur but is rare).
        """
        if len(stamps) == 0:
            return False, None, None
        ns   = ((stamps.day - 1) // 7 + 1).unique()
        dows = stamps.dayofweek.unique()
        if len(ns) == 1 and len(dows) == 1:
            return True, int(ns[0]), int(dows[0])
        return False, None, None


    def cluster_into_patterns(stamps: pd.DatetimeIndex,
                               source_pair: str | None = None) -> list[dict]:
        """Stage B — group spike timestamps into parametric patterns.

        Recognizers (apply in order; first match per (hh, mm) bucket wins):
          1. wom (week-of-month + weekday): e.g., 1st Friday of every month at 12:30 UTC -> NFP-shaped
          2. dow (weekly recurrence): e.g., every Wednesday at 18:00 UTC -> FOMC-presser-shaped
          3. mom (Nth day of every month): e.g., 15th at 14:00 UTC
          4. yearly (annual single-date): e.g., 12-25 -> Christmas thin liquidity
          5. dates (oneoff): unrecognized — preserved as raw timestamp list

        Each emitted pattern has:
            pattern: str (one of: wom, dow, mom, yearly, dates)
            time: str ("HH:MM") UTC
            duration_min: int (default 30 — D-10 ~30-min release windows)
            affects: list[str] (3-letter currency codes; default _pair_currencies(source_pair) if provided)
            source: str ("empirical")
            evidence_count: int (number of underlying detected timestamps)
            confidence: str (high if N>=20, medium if N>=8, low otherwise)
            ... pattern-specific fields (n, dow, day, month, dates) ...
        """
        if len(stamps) == 0:
            return []

        affects_default: list[str] = _pair_currencies(source_pair) if source_pair else []

        patterns: list[dict] = []
        # Group by (hour, minute) — recurring events fire at fixed UTC times
        by_time: dict[tuple[int, int], pd.DatetimeIndex] = {}
        for ts in stamps:
            key = (ts.hour, ts.minute)
            by_time.setdefault(key, []).append(ts)
        by_time = {k: pd.DatetimeIndex(v) for k, v in by_time.items()}

        for (hh, mm), group in by_time.items():
            # Skip noise — single-occurrence groups are oneoffs unless we hit a fallback
            time_str = f"{hh:02d}:{mm:02d}"
            n_evidence = len(group)
            confidence = "high" if n_evidence >= 20 else "medium" if n_evidence >= 8 else "low"
            base = {"time": time_str, "duration_min": 30,
                    "affects": list(affects_default), "source": "empirical",
                    "evidence_count": n_evidence, "confidence": confidence}

            # Try wom (week-of-month)
            ok, n, dow = _matches_wom(group)
            if ok and n is not None and dow is not None:
                patterns.append({"pattern": "wom", "n": n, "dow": dow, **base})
                continue
            # Try dow (every weekday at HH:MM)
            dows = group.dayofweek.unique()
            if len(dows) == 1:
                patterns.append({"pattern": "dow", "dow": int(dows[0]), **base})
                continue
            # Try mom (Nth day of every month)
            days = group.day.unique()
            if len(days) == 1:
                patterns.append({"pattern": "mom", "day": int(days[0]), **base})
                continue
            # Try yearly (single month-day)
            mds = pd.Index([(ts.month, ts.day) for ts in group]).unique()
            if len(mds) == 1:
                m, d = mds[0]
                patterns.append({"pattern": "yearly", "month": int(m), "day": int(d), **base})
                continue
            # Fallback: dates (oneoff list)
            patterns.append({"pattern": "dates",
                             "dates": [ts.strftime("%Y-%m-%d") for ts in group], **base})

        return patterns
    ```

    ```python
    # ── Risk calendar Stage C: YAML round-trip emission ──────────────────────

    def _merge_detected_with_manual(detected: list[dict],
                                     manual: list[dict]) -> list[dict]:
        """Per CONTEXT D-12: detection takes precedence on (pattern, time, affects) conflict.

        Conflict key = (pattern, time, sorted_affects_tuple). If a detected entry shares
        the conflict key with a manual entry, the detected entry wins (manual is dropped).
        Otherwise, both are preserved.

        Returns the merged list with detected entries first, then non-conflicting manual entries.
        """
        def _key(p: dict) -> tuple:
            affects = tuple(sorted(p.get("affects", []) or []))
            return (p.get("pattern"), p.get("time"), affects)

        detected_keys = {_key(p) for p in detected}
        kept_manual = [m for m in manual if _key(m) not in detected_keys]
        return list(detected) + kept_manual


    def write_risk_calendar(detected: list[dict], out_path: Path) -> None:
        """Stage C — emit ruamel.yaml round-trippable file; merge manual entries from existing file.

        Per CONTEXT D-11: parametric patterns, NOT calendar dates.
        Per CONTEXT D-12: manual entries (source: manual) preserved across re-runs;
                           detection takes precedence on conflicts; comments preserved.
        Per RESEARCH §Pitfall (PyYAML drops comments): use ruamel.yaml YAML(typ="rt") only.
        """
        from ruamel.yaml import YAML
        from ruamel.yaml.comments import CommentedMap, CommentedSeq

        yaml = YAML(typ="rt")
        yaml.default_flow_style = False
        yaml.indent(mapping=2, sequence=4, offset=2)

        # If the file exists, preserve operator manual entries (D-12)
        existing_data: CommentedMap | dict
        manual: list[dict] = []
        if out_path.exists():
            with out_path.open() as f:
                existing_data = yaml.load(f) or CommentedMap()
            existing_blackouts = existing_data.get("blackouts") or []
            manual = [b for b in existing_blackouts if b.get("source") == "manual"]
        else:
            existing_data = CommentedMap()

        # Merge — detection wins on conflict; non-conflicting manual entries preserved
        merged = _merge_detected_with_manual(detected, manual)

        # Update structure (preserves any non-blackout keys + top-level comments)
        existing_data["blackouts"] = merged

        # If this is a fresh file, add the header comment block
        if not out_path.exists() or "blackouts" not in (existing_data.ca.items if hasattr(existing_data, "ca") else {}):
            try:
                existing_data.yaml_set_comment_before_after_key(
                    "blackouts",
                    before=(
                        "Phase 8.5 risk calendar — empirically detected (D-09) + "
                        "operator-curated (D-12).\n"
                        "Re-run 'cd V2 && python -m scripts.run_temporal_analysis' to refresh "
                        "detected entries; manual entries (source: manual) are preserved "
                        "verbatim across runs.\n"
                        "Detection: realized range > 2.5σ above same-(hour-of-day, day-of-week) "
                        "4yr baseline (D-10).\n"
                    ),
                )
            except (AttributeError, TypeError):
                # Older ruamel versions or fresh dicts may not support comment API
                pass

        out_path.parent.mkdir(parents=True, exist_ok=True)
        with out_path.open("w") as f:
            yaml.dump(existing_data, f)


    def detect_and_write_risk_calendar(cache: OHLCVCache, end_ts: pd.Timestamp,
                                        out_path: Path,
                                        pairs: list[str] | None = None,
                                        timeframes: tuple[str, ...] = ("H1",)) -> Path:
        """Top-level pipeline (called from run_temporal_analysis CLI inside the existing PitClock).

        Iterates the active pair set across the given timeframes (default H1 only — adequate granularity
        per RESEARCH Open Question 1 recommendation), runs detect_blackout_bars + cluster_into_patterns
        for each, unions the detected timestamps, and emits the merged risk_calendar.yaml.
        """
        from v3_intelligence.pair_config import PAIR_CONFIGS
        if pairs is None:
            pairs = list(PAIR_CONFIGS.keys())

        all_patterns: list[dict] = []
        for pair in pairs:
            for tf in timeframes:
                try:
                    bars = cache.get_bars(pair, tf,
                                           start=end_ts - pd.Timedelta(days=4 * 366),
                                           end=end_ts)
                    if bars.empty:
                        continue
                    stamps = detect_blackout_bars(bars, sigma=DETECTION_SIGMA)
                    if len(stamps) == 0:
                        continue
                    patterns = cluster_into_patterns(stamps, source_pair=pair)
                    all_patterns.extend(patterns)
                except Exception:
                    # Cache miss or failover — skip this (pair, tf) but continue batch
                    continue

        write_risk_calendar(all_patterns, out_path)
        return out_path
    ```
