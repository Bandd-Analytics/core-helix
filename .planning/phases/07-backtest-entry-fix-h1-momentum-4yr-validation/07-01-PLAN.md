     - Swing, M15, H1 scalp, and momentum loops use next-bar open as entry_price

   def _synthetic_h1(n: int = 300, seed: int = 42) -> pd.DataFrame:
       """Build a synthetic H1 DataFrame with ATR, Z-score, daily_z columns.

       Upward-drifting close so next-bar open < current-bar close on average
       → biased entry (close fill) out-performs corrected entry (next open).
       """
       rng = np.random.default_rng(seed)
       idx = pd.date_range("2024-01-01", periods=n, freq="h")
       close = 100 + np.cumsum(rng.normal(0.02, 0.1, n))
       open_ = np.r_[close[0], close[:-1] + rng.normal(0, 0.05, n - 1)]
       df = pd.DataFrame({
           "Open": open_, "High": close + 0.1, "Low": close - 0.1, "Close": close,
           "atr": np.full(n, 0.5),
           "z_score": rng.normal(0, 1, n),
           "daily_z": rng.normal(0, 1, n),
       }, index=idx)
       return df
